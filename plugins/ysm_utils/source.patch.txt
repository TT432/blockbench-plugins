+// Local compatibility revision for YSM Utils 2.5.2 on Blockbench Community 5.3.74.
+// Keep the original plugin's behavior without exposing native modules globally.
+export const ysmCommunityNative = (() => {
+    const legacy = typeof electron === 'undefined' ? null : electron;
+    const requestModule = typeof requireNativeModule === 'function'
+        ? requireNativeModule
+        : require;
+    const modules = Object.create(null);
+
+    function nativeModule(name) {
+        if (legacy && legacy[name]) return legacy[name];
+        if (modules[name]) return modules[name];
+        const api = requestModule(name, {
+            optional: true,
+            message: name === 'dialog'
+                ? 'YSM uses native confirmation dialogs before saving or overwriting files.'
+                : 'YSM opens model folders and moves removed model assets to the recycle bin.'
+        });
+        if (!api) {
+            const message = 'YSM: native ' + name + ' access was not granted; the operation was canceled.';
+            Blockbench.showQuickMessage(message, 5000);
+            throw new Error(message);
+        }
+        modules[name] = api;
+        return api;
+    }
+
+    return {
+        loadAnimationFile(file) {
+            const codec = typeof AnimationCodec !== 'undefined' && AnimationCodec.codecs
+                ? AnimationCodec.codecs.bedrock
+                : null;
+            return codec ? codec.loadFile(file) : Animator.loadFile(file);
+        },
+        dialog: {
+            async showOpenDialog(options) {
+                if (legacy && legacy.dialog) return legacy.dialog.showOpenDialog(options);
+                const properties = options.properties || [];
+                let filePaths = [];
+                if (properties.includes('openDirectory')) {
+                    const directory = Blockbench.pickDirectory({
+                        title: options.title,
+                        startpath: options.defaultPath
+                    });
+                    if (directory) filePaths.push(directory);
+                } else {
+                    const filters = options.filters || [];
+                    // Desktop import with readtype:none calls back synchronously,
+                    // and does not read or import any contents into the project.
+                    Blockbench.import({
+                        title: options.title,
+                        startpath: options.defaultPath,
+                        type: filters.map(filter => filter.name).join(', ') || 'Files',
+                        extensions: [...new Set(filters.flatMap(filter => filter.extensions))],
+                        multiple: properties.includes('multiSelections'),
+                        readtype: 'none'
+                    }, files => {
+                        filePaths = files.map(file => file.path);
+                    });
+                }
+                return {canceled: filePaths.length === 0, filePaths};
+            },
+            showMessageBoxSync(options) {
+                // The scoped dialog module binds the app window itself.
+                return nativeModule('dialog').showMessageBoxSync(options);
+            }
+        },
+        shell: {
+            openPath(filePath) {
+                return nativeModule('shell').openPath(filePath);
+            },
+            trashItem(filePath) {
+                return nativeModule('shell').trashItem(filePath);
+            }
+        }
+    };
+})();
